# CabinWatch agent - install on each machine that has a camera attached.
#
#   pip install -r agent/requirements.txt

# Pinned below 5.0 on purpose. OpenCV 5 dropped HOGDescriptor and
# CascadeClassifier from the Python bindings, which is what person detection
# uses to tell a person from a moving curtain without downloading a model.
# On 5.x the cabin camera silently falls back to alerting on any movement.
opencv-python>=4.9.0,<5.0
numpy>=1.24.0
websocket-client>=1.7.0

# Puts the sound into the saved recordings. OpenCV writes video only, so
# ffmpeg joins each clip to the audio captured alongside it. This package
# carries its own ffmpeg, so nothing has to be installed separately.
imageio-ffmpeg>=0.5.1

# Microphone and speaker, for baby-sound alerts and two-way talk.
# Drop this line if a camera has no mic and you do not need to talk back.
sounddevice>=0.4.6

# The tray icon on Windows and macOS. Without these the agent still runs -
# it prints to the terminal as it always did - but the packaged Windows
# build has no console at all, so there would be nothing on screen and no
# way to reach the settings or stop it.
pystray>=0.19.5
pillow>=10.0.0

# Only needed for an ONVIF PTZ camera (most rotating IP cameras).
# onvif-zeep>=0.2.12

# Only needed for the "http" PTZ backend.
# requests>=2.31.0
